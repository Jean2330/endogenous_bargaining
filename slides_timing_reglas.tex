\begin{frame}{Piezas del juego}
    \begin{wideitemize}
        \item \textbf{Jugadores}: principal neutral al riesgo (él) y agente con utilidad CARA (ella)
        \item \textbf{Tiempo}: discreto, $t=0,1,2,\dots$, horizonte infinito, descuento común $\rho$
        \item \textbf{Estado}: riqueza acumulada de la agente, $W_t$, información pública
        \item \textbf{Contrato}: lineal, $w_t=\alpha_t+\beta_t y_t$, dura un solo periodo
        \item \textbf{Tecnología}: $y_t=\theta a_t+\sigma\varepsilon_t$, con $\varepsilon_t\sim N(0,1)$ i.i.d. y costo de esfuerzo $\tfrac{k}{2}a_t^2$
        \item \textbf{Responsabilidad limitada (LL)}: $\alpha_t+\beta_t\underline{y}\ge\underline{w}(W_t)=\overline{w}-\gamma_w W_t$. Mientras más rica la agente, más laxa la restricción
        \item \textbf{Reparto}: Nash bargaining con peso exógeno $\delta$ para la agente y pagos de desacuerdo exógenos $(\overline{\Pi},\overline{\CE})$
    \end{wideitemize}
\end{frame}

\begin{frame}{¿Qué se negocia en $t$?}
    \begin{wideitemize}
        \item Se negocia $(\alpha_t,\beta_t)$, el contrato que rige la producción y el pago del mismo periodo
        \item El contrato $(\alpha_{t+1},\beta_{t+1})$ se negocia desde cero al inicio de $t+1$, con $W_{t+1}$ como estado
        \item El futuro entra solo por la continuación: $(\alpha_t,\beta_t)$ mueve $W_{t+1}$, y $W_{t+1}$ determina el piso de LL de mañana y con él el resultado de la negociación de $t+1$
        \item Las partes saben cómo se resolverá mañana, pero no pueden fijar sus términos hoy
    \end{wideitemize}
    \vspace{0.3cm}
    \[
    \max_{(\alpha_t,\beta_t)\in\Gamma(W_t)}\;
    \Big[\Pi_t+\rho\,E\,V_P(W_{t+1})-\overline{\Pi}\Big]^{1-\delta}
    \Big[\CE_t+\rho\,E\,V_A(W_{t+1})-\overline{\CE}\Big]^{\delta}
    \]
    \begin{center}
    {\footnotesize con $\Pi_t=D(\beta_t)-\alpha_t$, $\;\CE_t=\alpha_t+C(\beta_t)$, y $\Gamma(W_t)$ el conjunto de contratos que cumplen LL, IC e IR}
    \end{center}
\end{frame}

\begin{frame}{Dos regímenes dentro del Nash}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{LL holgada} ($W_t\ge\overline{W}$)
            \begin{wideitemize}
                \item $\alpha$ es una transferencia pura: la frontera de Pareto es una recta de pendiente $-1$
                \item $\beta=\beta^{\eff}=\dfrac{\theta^2/k}{\gamma\sigma^2+\theta^2/k}$, constante en $t$
                \item $\alpha$ reparte el excedente
                \item La agente recibe exactamente la fracción $\delta$: $\lambda_t=\delta$
            \end{wideitemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{LL activa} ($W_t<\overline{W}$)
            \begin{wideitemize}
                \item La LL ata $\alpha_t=\underline{w}(W_t)-\beta_t\underline{y}$, así que $\alpha$ deja de ser libre
                \item Solo queda $\beta$ para elegir: $\beta^*(W_t)\ne\beta^{\eff}$
                \item $\beta$ cumple dos funciones a la vez (incentivos y reparto) y se pierde excedente
                \item La frontera se curva y $\lambda_t\ne\delta$
            \end{wideitemize}
        \end{column}
    \end{columns}
    \vspace{0.3cm}
    \begin{center}
    {\footnotesize $\overline{W}$: riqueza a la que el $\alpha$ del régimen holgado cumple la LL con igualdad}
    \end{center}
\end{frame}

\begin{frame}{¿Dónde entra el poder de negociación?}
    \begin{wideitemize}
        \item $\delta$ \textbf{(primitivo, exógeno)}: el peso de la agente en el producto de Nash. Constante en el tiempo, no se actualiza
        \item $\lambda_t$ \textbf{(realizado, endógeno)}: la fracción del excedente neto que efectivamente recibe la agente
    \end{wideitemize}
    {\small
    \[
    \lambda_t=\frac{\CE_t-\overline{\CE}}{\Sigma_t},\qquad
    \Sigma_t=(\CE_t-\overline{\CE})+(\Pi_t-\overline{\Pi})
    \]}
    \begin{wideitemize}
        \item LL holgada: $\lambda_t=\delta$. LL activa: $\lambda_t\ne\delta$ (en la calibración base $\lambda_t>\delta$ y decrece con $W_t$ hasta $\delta$ en $\overline{W}$)
        \item Como $W$ se mueve con los shocks de producto, $\lambda_t$ evoluciona con el historial de shocks, por la cadena
    \end{wideitemize}
    {\small
    \[
    W_t\;\to\;\underline{w}(W_t)\;\to\;\Gamma(W_t)\;\to\;(\alpha_t,\beta_t)\text{ con el mismo }\delta\;\to\;\lambda_t=\lambda(W_t)
    \]}
\end{frame}

% =========================================================
\begin{frame}{Qué se actualiza entre periodos}
    \begin{wideitemize}
        \item Una sola cosa: la riqueza, $W_{t+1}=W_t+\alpha_t+\beta_t y_t-\tfrac{k}{2}a_t^2$. La agente no consume: toda la paga neta del costo de esfuerzo se acumula
        \item Todo lo demás es función de $W_t$: $\alpha^*(W)$, $\beta^*(W)$, $\lambda(W)$, $V_P(W)$, $V_A(W)$
        \item No se actualizan $\delta$, los pagos de desacuerdo, $\underline{y}$ ni los parámetros de la tecnología
        \item Lazo de retroalimentación: $W_t\to\underline{w}(W_t)\to\beta^*_t\to\lambda_t$, y de vuelta por $\beta^*_t\sigma\varepsilon_t\to W_{t+1}$
    \end{wideitemize}
    \textbf{Notación propuesta para $\beta$}
    \begin{wideitemize}
        \item $\beta^{\eff}$ (sin subíndice) para el nivel constante del benchmark y del régimen holgado
        \item $\beta^*(W_t)$ para el régimen restringido, donde $t$ entra solo a través de $W_t$
        \item En la parte estática del borrador $\beta^*_t$ no varía con $t$, aunque el subíndice lo sugiere
    \end{wideitemize}
\end{frame}
